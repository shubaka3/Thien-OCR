from fastapi import FastAPI, UploadFile, File, HTTPException, Form
from fastapi.responses import JSONResponse
from paddleocr import PaddleOCR
from PIL import Image
import pytesseract
import io
import numpy as np
import time

app = FastAPI()

ocr_cache = {}

def get_paddle_ocr(lang: str):
    if lang not in ocr_cache:
        ocr_cache[lang] = PaddleOCR(use_angle_cls=True, lang=lang)
    return ocr_cache[lang]

def read_image(contents: bytes):
    image = Image.open(io.BytesIO(contents)).convert("RGB")
    return np.array(image)

@app.post("/ocr-basic")
async def ocr_basic(
    file: UploadFile = File(...),
    lang: str = Form("en"),
    time_return: bool = Form(False)
):
    try:
        start = time.time()
        contents = await file.read()
        image_np = read_image(contents)
    except Exception:
        raise HTTPException(status_code=400, detail="Invalid image file")

    ocr = get_paddle_ocr(lang)
    result = ocr.ocr(image_np, cls=True)

    output = []
    for line in result:
        output.append({
            "box": line[0],
            "text": line[1][0],
            "confidence": line[1][1]
        })

    resp = {"result": output}
    if time_return:
        resp["time_ms"] = round((time.time() - start) * 1000, 2)

    return JSONResponse(content=resp)

@app.post("/ocr-full-multilang")
async def ocr_full_multilang(
    file: UploadFile = File(...),
    lang: str = Form("vi")
):
    try:
        contents = await file.read()
        image = Image.open(io.BytesIO(contents)).convert("RGB")
        image_np = np.array(image)
    except Exception:
        raise HTTPException(status_code=400, detail="Invalid image file")

    result = get_paddle_ocr(lang).ocr(image_np, cls=True)

    output = []

    if len(result) > 0:
        for line in result[0]:
            box, (_, _) = line

            if not isinstance(box, list) or not all(isinstance(pt, list) and len(pt) == 2 for pt in box):
                continue

            try:
                x_coords = [int(pt[0]) for pt in box]
                y_coords = [int(pt[1]) for pt in box]
            except Exception:
                continue

            x_min, x_max = max(0, min(x_coords)), max(x_coords)
            y_min, y_max = max(0, min(y_coords)), max(y_coords)

            cropped = image_np[y_min:y_max, x_min:x_max]

            try:
                tesseract_lang = "vie" if lang == "vi" else "eng"
                text = pytesseract.image_to_string(cropped, lang=tesseract_lang)
            except Exception as e:
                text = f"[ERROR: {str(e)}]"

            output.append({
                "box": box,
                "text": text.strip()
            })

    return JSONResponse(content={"result": output})
